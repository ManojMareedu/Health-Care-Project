# Dependencies for the Streamlit Community Cloud deployment only.
#
# Deliberately narrower than requirements.txt. The hosted dashboard loads the
# pickled sklearn pipeline from exported_model/ and scores it -- it never trains,
# never writes to a tracking server, and never serves HTTP. So mlflow, zenml,
# dvc, evidently, fastapi, and uvicorn are all absent.
#
# EVERY FLOOR BELOW EXISTS FOR ONE REASON: cp314 wheel availability.
#
# Streamlit Community Cloud currently builds on Python 3.14 and ignores
# runtime.txt (streamlit/streamlit#15326). Treat 3.14 as the deploy environment
# regardless of what runtime.txt says. Any package here without a cp314 wheel is
# built from source, and the build image has no cmake or C/Fortran toolchain, so
# the install dies -- which is exactly how the first two deploys failed.
#
# Verified minimum versions shipping cp314 wheels (checked against PyPI):
#   numpy        2.3.2   -- no numpy 1.x release has one at all
#   pandas       2.3.3
#   scikit-learn 1.7.2
#   scipy        1.16.1  (transitive)
#   shap         0.50.0
#   pyarrow      22.0.0  (transitive via streamlit, pinned here so the resolver
#                         cannot land on an older one; streamlit requires >=7)
#   xgboost      any -- its wheels are py3-none-<platform>, version agnostic
#
# Before changing any pin, confirm the new version has a cp314 wheel.
streamlit>=1.38,<2.0
pandas>=2.3.3,<3.0
numpy>=2.3.2,<3.0
# Exact pin: must match the version the pipelines in exported_model/ were
# pickled with. See exported_model/*/MLmodel. Changing it requires retraining.
scikit-learn==1.9.0
# <3.0 avoids xgboost 3.x pulling nvidia-nccl-cu12 (~200MB) into a CPU-only demo.
xgboost>=2.1,<3.0
shap>=0.50,<1.0
joblib>=1.3,<2.0
pydantic>=2.9,<3.0
pyarrow>=22.0
